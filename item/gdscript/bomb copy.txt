extends Area2D

@onready var sprite: AnimatedSprite2D = $AnimatedSprite2D
@onready var audio: AudioStreamPlayer = $AudioStreamPlayer

## 虚假值: 为true时，不会爆炸
@export var fake: bool = false:
	set(v):
		if (not fakable) and is_node_ready():
			return
		if fake == v:
			return
		fake = v
		if is_node_ready():
			_update_fake(0)
		

## 可造假值: 为true时，虚假值可被更改
@export var fakable: bool = false

## 爆炸后等待多久重开（秒）
@export var restart_delay: float = 1.0

func _ready() -> void:
	body_entered.connect(_on_body_entered)
	_update_fake(0)

func _on_body_entered(body: Node2D) -> void:
	if not body.is_in_group("player"):
		return
	if not fake:
		fakable = false
		sprite.play("explode")
	
		audio.pitch_scale = randf_range(0.8, 1.4)
		audio.play()
	
		# 等待并重载当前场景
		await get_tree().create_timer(restart_delay).timeout
		TransitionManager.reload_current_scene()


func _update_fake(alpha: float = 0.5) -> void:
	if not self is CanvasItem:
		return 
	if fake:
		self.modulate.a = alpha
	else:
		self.modulate.a = 1.0
