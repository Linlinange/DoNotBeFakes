extends Node2D

## 相机缩放: 这将影响CanvasLayer的缩放，以及字体的实际大小和实际位置
@export var zoom: Vector2 = Vector2(1,1)
## 文本内容
@export var text: String = "请输入文本"
## 文本默认可见性
@export var default_visible: bool = true
## 文本偏移
@export var label_offset: Vector2 = Vector2(0, 0)
## 文字和气泡边框的间距
@export var padding: Vector2 = Vector2(10, 6)
## 气泡宽度兼文字最大宽度，超过自动换行（填 0 表示不换行且气泡跟随文字边框大小）
#@export var width: float = 180.0
## 气泡整体偏移（相对于 chat 原点，默认向上飘在头顶）
@export var bubble_offset: Vector2 = Vector2(0, -32)

@onready var bubble: NinePatchRect = $NinePatchRect
@onready var layer: CanvasLayer = $CanvasLayer
@onready var label: Label = $CanvasLayer/Label

# Called when the node enters the scene tree for the first time.
func _ready() -> void:
	set_canvas_layer(zoom)
	set_text_content(text)
	set_text_visible(default_visible)
	_resize()
	return

# ========== 私有方法 ==========
func _resize() -> void:
	label.reset_size()  # 立即根据文本重算尺寸，不用等帧
	print(label.size)
	
	var content_size = label.size
	bubble.size.x = content_size.x/self.zoom.x + padding.x*2
	bubble.size.y = content_size.y/self.zoom.y + padding.y*2
	
	# 气泡底部中心对准原点（父节点头顶）
	self.position = Vector2(-bubble.size.x / 2, -bubble.size.y) + bubble_offset
	_set_text_position(self.global_position)


## 设置文本位置
func _set_text_position(new_pos: Vector2) -> void:
	label.global_position.x = (new_pos + padding).x/layer.scale.x + label_offset.x
	label.global_position.y = (new_pos + padding).y/layer.scale.y + label_offset.y
	return
# ========== 公共方法 ==========

func say(text: String, duration: float = -1.0) -> void:
	show_text(text)

func show_text(text: String) -> void:
	label.text = text
	_resize()

## 设置CanvasLayer: new_zoom表示当前正在使用的相机缩放
func set_canvas_layer(new_zoom: Vector2) -> void:
	self.zoom = new_zoom
	layer.scale.x = 1/self.zoom.x
	layer.scale.y = 1/self.zoom.y
	return

## 设置文本内容
func set_text_content(content: String) -> void:
	self.text = content
	label.text = self.text

## 设置文本可见性
func set_text_visible(visibility: bool) -> void:
	label.visible = visibility
	return

## 切换文本可见性
func toggle_text_visible() -> void:
	label.visible = !label.visible
	return

## 返回 action 绑定的第一个键盘按键的显示名称
func get_key_name(action: String) -> String:
	var events := InputMap.action_get_events(action)
	for ev in events:
		if ev is InputEventKey:
			# Godot 4 返回如 "W", "Space", "Shift"
			return ev.as_text()
		if ev is InputEventMouseButton:
			return "鼠标按键 %d" % ev.button_index
	return "?"
